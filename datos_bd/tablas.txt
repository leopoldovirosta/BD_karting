CREATE TABLE regiones (
    id_region INT UNSIGNED NOT NULL AUTO_INCREMENT PRIMARY KEY,
    nombre_region VARCHAR(30) COLLATE utf8mb4_spanish_ci NOT NULL
    ) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_spanish_ci;

CREATE TABLE areas (
    id_area INT UNSIGNED NOT NULL AUTO_INCREMENT PRIMARY KEY,
    nombre_area VARCHAR(30) COLLATE utf8mb4_spanish_ci NOT NULL
    ) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_spanish_ci;

CREATE TABLE escuderias (
    id_escuderia INT UNSIGNED NOT NULL AUTO_INCREMENT PRIMARY KEY,
    nombre_escuderia VARCHAR(50) COLLATE utf8mb4_spanish_ci NOT NULL,
    logo_escuderia VARCHAR(255) COLLATE utf8mb4_spanish_ci NULL,
    web_escuderia VARCHAR(100) COLLATE utf8mb4_spanish_ci NULL
    ) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_spanish_ci;

CREATE TABLE patrocinadores (
    id_patrocinador INT UNSIGNED NOT NULL AUTO_INCREMENT PRIMARY KEY,
    nombre_patrocinador VARCHAR(50) COLLATE utf8mb4_spanish_ci NOT NULL,
    logo_patrocinador VARCHAR(255) COLLATE utf8mb4_spanish_ci NULL,
    web_patrocinador VARCHAR(100) COLLATE utf8mb4_spanish_ci NULL
    ) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_spanish_ci;

CREATE TABLE categorias (
    id_categoria INT UNSIGNED NOT NULL AUTO_INCREMENT PRIMARY KEY,
    nombre_categoria VARCHAR(30) COLLATE utf8mb4_spanish_ci NOT NULL
    ) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_spanish_ci;

CREATE TABLE campeonatos (
    id_cto INT UNSIGNED NOT NULL AUTO_INCREMENT PRIMARY KEY,
    nombre_cto VARCHAR(50) COLLATE utf8mb4_spanish_ci NOT NULL,
    logo_cto VARCHAR(255) COLLATE utf8mb4_spanish_ci NULL
    ) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_spanish_ci;

CREATE TABLE marcas (
    id_marca INT UNSIGNED NOT NULL AUTO_INCREMENT PRIMARY KEY,
    nombre_marca VARCHAR(100) COLLATE utf8mb4_spanish_ci NOT NULL,
    pais_origen VARCHAR(50) COLLATE utf8mb4_spanish_ci NOT NULL,
    logo_marca VARCHAR(255) COLLATE utf8mb4_spanish_ci NOT NULL,
    pagina_web VARCHAR(100) COLLATE utf8mb4_spanish_ci NOT NULL,
    es_marca_chasis TINYINT(1) DEFAULT 0,
    es_marca_motor TINYINT(1) DEFAULT 0,
    es_marca_ruedas TINYINT(1) DEFAULT 0
    ) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_spanish_ci;

CREATE TABLE chasis (
    id_chasis INT UNSIGNED NOT NULL AUTO_INCREMENT PRIMARY KEY,
    modelo_chasis VARCHAR(30) COLLATE utf8mb4_spanish_ci NOT NULL,
    material VARCHAR(255) COLLATE utf8mb4_spanish_ci NOT NULL,
    tubo_diametro VARCHAR(255) COLLATE utf8mb4_spanish_ci NULL,
    distancia_ejes VARCHAR(255) COLLATE utf8mb4_spanish_ci NULL,
    eje_trasero VARCHAR(255) COLLATE utf8mb4_spanish_ci NULL,
    sistema_frenado VARCHAR(255) COLLATE utf8mb4_spanish_ci NULL,
    categoria_objetivo VARCHAR(30) COLLATE utf8mb4_spanish_ci NULL,
    ano YEAR,
    homologacion VARCHAR(15) COLLATE utf8mb4_spanish_ci NULL,
    foto_chasis VARCHAR(255) COLLATE utf8mb4_spanish_ci NULL,
    id_marca INT UNSIGNED NOT NULL,
    CONSTRAINT fk_marca FOREIGN KEY (id_marca) REFERENCES marcas(id_marca)
    ) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_spanish_ci;

CREATE TABLE motores (
    id_motor INT UNSIGNED NOT NULL AUTO_INCREMENT PRIMARY KEY,
    modelo VARCHAR(30) COLLATE utf8mb4_spanish_ci NOT NULL,
    descripcion VARCHAR(30) COLLATE utf8mb4_spanish_ci NOT NULL DEFAULT 'Monocilíndrico de 2 tiempos',
    clase VARCHAR(30) COLLATE utf8mb4_spanish_ci NOT NULL,
    diametro DECIMAL(4,2) UNSIGNED NULL, 
    max_diametro DECIMAL(4,2) UNSIGNED NULL, 
    carrera DECIMAL(4,2) UNSIGNED NULL, 
    cilindrada DECIMAL(5,2) UNSIGNED NULL, 
    admision ENUM('Caja de laminas','Valvula rotativa','Admision por piston') DEFAULT 'Valvula de laminas',
    lubricacion VARCHAR(30) COLLATE utf8mb4_spanish_ci NOT NULL DEFAULT 'Mezcla de aceite',
    lumbreras TINYINT UNSIGNED NULL,
    encendido VARCHAR(50) COLLATE utf8mb4_spanish_ci NULL,
    regimen_max_rpm INT UNSIGNED NULL,
    transmision VARCHAR(50) COLLATE utf8mb4_spanish_ci NULL,
    refrigeracion ENUM('Aire', 'Agua') DEFAULT 'Agua',
    starter VARCHAR(30) COLLATE utf8mb4_spanish_ci NULL,
    homologacion VARCHAR(10) COLLATE utf8mb4_spanish_ci NULL,
    peso DECIMAL(4,2) UNSIGNED NULL, 
    carburador VARCHAR(50) COLLATE utf8mb4_spanish_ci NULL,
    potencia_max DECIMAL(4,2) UNSIGNED NULL,
    par_motor_max DECIMAL(4,2) UNSIGNED NULL,
    foto_motor VARCHAR(255) COLLATE utf8mb4_spanish_ci NULL,
    id_marca INT UNSIGNED NOT NULL,
    CONSTRAINT fk_marca FOREIGN KEY (id_marca) REFERENCES marcas(id_marca)
    ) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_spanish_ci;

CREATE TABLE ruedas (
    id_rueda INT UNSIGNED NOT NULL AUTO_INCREMENT PRIMARY KEY,
    modelo_rueda VARCHAR(30) COLLATE utf8mb4_spanish_ci NOT NULL,
    tam_delante VARCHAR(20) COLLATE utf8mb4_spanish_ci NULL,
    tam_atras VARCHAR(20) COLLATE utf8mb4_spanish_ci NULL,
    max_velocidad TINYINT UNSIGNED NULL,
    id_marca INT UNSIGNED NOT NULL,
    CONSTRAINT fk_marca FOREIGN KEY (id_marca) REFERENCES marcas(id_marca)
    ) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_spanish_ci;

CREATE TABLE pilotos (
    id_piloto INT UNSIGNED NOT NULL AUTO_INCREMENT PRIMARY KEY,
    nombre_piloto VARCHAR(30) COLLATE utf8mb4_spanish_ci NOT NULL,
    apellido_piloto VARCHAR(50) COLLATE utf8mb4_spanish_ci NOT NULL,
    fecha_nacimiento DATE NULL,
    web_piloto VARCHAR(100) COLLATE utf8mb4_spanish_ci NULL,
    email_piloto VARCHAR(60) COLLATE utf8mb4_spanish_ci NULL,
    foto_piloto VARCHAR(255) COLLATE utf8mb4_spanish_ci NOT NULL DEFAULT 'default.jpg',
    id_region INT UNSIGNED NOT NULL DEFAULT 1,
    id_escuderia INT UNSIGNED NOT NULL DEFAULT 1,
    id_patrocinador INT UNSIGNED NOT NULL DEFAULT 1,
    CONSTRAINT fk_region FOREIGN KEY (id_region) REFERENCES regiones(id_region),
    CONSTRAINT fk_escuderia FOREIGN KEY (id_escuderia) REFERENCES escuderias(id_escuderia),
    CONSTRAINT fk_patrocinador FOREIGN KEY (id_patrocinador) REFERENCES patrocinadores(id_patrocinador)
    ) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_spanish_ci;

CREATE TABLE circuitos (
    id_circuito INT UNSIGNED NOT NULL AUTO_INCREMENT PRIMARY KEY,
    nombre_circuito VARCHAR(50) COLLATE utf8mb4_spanish_ci NOT NULL,
    logo_circuito VARCHAR(255) COLLATE utf8mb4_spanish_ci NULL,
    web_circuito VARCHAR(100) COLLATE utf8mb4_spanish_ci NULL,
    direccion_circuito VARCHAR(100) COLLATE utf8mb4_spanish_ci NULL,
    localidad_circuito VARCHAR(30) COLLATE utf8mb4_spanish_ci NULL,
    telefono_circuito VARCHAR(15) COLLATE utf8mb4_spanish_ci NOT NULL,
    id_area INT UNSIGNED NOT NULL,
    altitud SMALLINT NOT NULL,
    longitud SMALLINT UNSIGNED NOT NULL,
    curvasizd TINYINT UNSIGNED NOT NULL,
    curvasdcha TINYINT UNSIGNED NOT NULL,
    velocidadmax TINYINT UNSIGNED NOT NULL,
    silueta VARCHAR(255) COLLATE utf8mb4_spanish_ci NOT NULL,
    CONSTRAINT fk_area FOREIGN KEY (id_area) REFERENCES areas(id_area)
    ) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_spanish_ci;

CREATE TABLE carreras_tipo (
    id_carrera_tipo INT UNSIGNED NOT NULL AUTO_INCREMENT PRIMARY KEY,
    nombre_carrera_tipo VARCHAR(15) COLLATE utf8mb4_spanish_ci NOT NULL
    ) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_spanish_ci;

CREATE TABLE carreras (
    id_carrera INT UNSIGNED NOT NULL AUTO_INCREMENT PRIMARY KEY,
    fecha_carrera DATE NOT NULL,
    dia ENUM('Despejado','Parcialmente nublado','Nublado') COLLATE utf8mb4_spanish_ci NOT NULL DEFAULT 'Despejado',
    num_vueltas TINYINT UNSIGNED NULL,
    pista ENUM('Seco','Mojado') COLLATE utf8mb4_spanish_ci NOT NULL DEFAULT 'Seco',
    temperatura TINYINT NULL,
    humedad TINYINT UNSIGNED NULL,
    presion SMALLINT UNSIGNED NULL,
    viento TINYINT UNSIGNED NULL,
    orientacion ENUM('Norte','Sur','Este','Oeste','Noreste','Noroeste','Sureste','Suroeste','Oeste/Noroeste','Norte/Noroeste','Norte/Noreste','Este/Noreste','Este/Sureste','Sur/Sureste','Sur/Suroeste','Oeste/Suroeste') COLLATE utf8mb4_spanish_ci NOT NULL,
    tasfalto TINYINT NULL,
    id_carrera_tipo INT UNSIGNED NOT NULL,
    id_circuito INT UNSIGNED NOT NULL,
    CONSTRAINT fk_carrera_tipo FOREIGN KEY (carrera_tipo_id) REFERENCES carreras_tipo(id_carrera_tipo), 
    CONSTRAINT fk_circuito FOREIGN KEY (circuito_id) REFERENCES circuitos(id_circuito)
    ) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_spanish_ci;

CREATE TABLE campeonatos_carreras (
    id_cto INT UNSIGNED NOT NULL,
    id_carrera INT UNSIGNED NOT NULL,
    PRIMARY KEY (cto_id, carrera_id),
    CONSTRAINT fk_rel_cto FOREIGN KEY (cto_id) REFERENCES campeonatos(id_cto),
    CONSTRAINT fk_rel_carrera FOREIGN KEY (carrera_id) REFERENCES carreras(id_carrera)
    ) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_spanish_ci;

CREATE TABLE resultados (
    id_resultado INT UNSIGNED NOT NULL AUTO_INCREMENT,
    id_carrera INT UNSIGNED NOT NULL,
    id_piloto INT UNSIGNED NOT NULL,
    id_categoria INT UNSIGNED NOT NULL,
    dorsal SMALLINT,
    posicion TINYINT UNSIGNED NOT NULL,
    comentario_posicion ENUM('Normal', 'DNF', 'DSQ', 'DNS', 'EXC') COLLATE utf8mb4_spanish_ci DEFAULT 'Normal',
    tiempo_total TIME(3) NULL,
    mejor_vuelta TIME(3) NULL,
    num_vueltas_completadas TINYINT UNSIGNED,
    id_chasis INT UNSIGNED NULL,
    id_motor INT UNSIGNED NULL,
    id_rueda INT UNSIGNED NULL,
    PRIMARY KEY (id_resultado),
    INDEX idx_resultado_carrera (id_carrera),
    INDEX idx_resultado_piloto (id_piloto),
    CONSTRAINT fk_res_carrera FOREIGN KEY (id_carrera) REFERENCES carreras(id_carrera),
    CONSTRAINT fk_res_piloto FOREIGN KEY (id_piloto) REFERENCES pilotos(id_piloto),
    CONSTRAINT fk_res_categoria FOREIGN KEY (id_categoria) REFERENCES categorias(id_categoria),
    CONSTRAINT fk_res_chasis FOREIGN KEY (id_chasis) REFERENCES chasis(id_chasis),
    CONSTRAINT fk_res_motor FOREIGN KEY (id_motor) REFERENCES motores(id_motor),
    CONSTRAINT fk_res_rueda FOREIGN KEY (id_rueda) REFERENCES ruedas(id_rueda)
    ) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_spanish_ci;

CREATE TABLE sistemas_puntuacion (
    id_cto INT UNSIGNED NOT NULL,
    id_tipo_carrera INT UNSIGNED NOT NULL,
    posicion TINYINT UNSIGNED NOT NULL,
    puntos_asignados TINYINT UNSIGNED NULL,
    PRIMARY KEY (id_cto, id_tipo_carrera, posicion),
    CONSTRAINT fk_sp_campeonato FOREIGN KEY (id_cto) REFERENCES campeonatos (id_cto) ON DELETE CASCADE,
    CONSTRAINT fk_sp_tipo_carrera FOREIGN KEY (id_tipo_carrera) REFERENCES carreras_tipo(id_carrera_tipo) ON DELETE CASCADE
    ) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_spanish_ci;

CREATE TABLE campeonatos_categorias (
    id_cto INT UNSIGNED NOT NULL,
    id_categoria INT UNSIGNED NOT NULL,
    PRIMARY KEY (id_cto, id_categoria),
    CONSTRAINT fk_cto_rel FOREIGN KEY (id_cto) REFERENCES campeonatos(id_cto) ON DELETE CASCADE,
    CONSTRAINT fk_cat_rel FOREIGN KEY (id_categoria) REFERENCES categorias(id_categoria) ON DELETE CASCADE
    ) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_spanish_ci;
