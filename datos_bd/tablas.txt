CREATE TABLE regiones (
    id_region INT UNSIGNED NOT NULL AUTO_INCREMENT PRIMARY KEY,
    nombre_region ENUM('Privado','Andalucía','Aragón','Asturias','Cantabria','Castilla La Mancha','Castilla y León','Cataluña','Ceuta','Extremadura','Galicia','Islas Canarias','Islas Baleares','La Rioja','Madrid','Melilla','Murcia','Navarra','País Vasco','Valencia') COLLATE utf8mb4_spanish_ci NOT NULL)  ENGINE=InnoDB  DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_spanish_ci;

CREATE TABLE escuderias (
    id_escuderia INT UNSIGNED NOT NULL AUTO_INCREMENT PRIMARY KEY,
    nombre_escuderia VARCHAR(50) COLLATE utf8mb4_spanish_ci NOT NULL,
    web_escuderia VARCHAR(100) COLLATE utf8mb4_spanish_ci,
    UNIQUE (nombre_escuderia)) ENGINE=InnoDB  DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_spanish_ci;

CREATE TABLE patrocinadores (
    id_patrocinador INT UNSIGNED NOT NULL AUTO_INCREMENT PRIMARY KEY,
    nombre_patrocinador VARCHAR(50) COLLATE utf8mb4_spanish_ci NOT NULL,
    web_patrocinador VARCHAR(100) COLLATE utf8mb4_spanish_ci,
    UNIQUE (nombre_patrocinador))  ENGINE=InnoDB  DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_spanish_ci;

CREATE TABLE chasis (
    id_chasis INT UNSIGNED NOT NULL AUTO_INCREMENT PRIMARY KEY,
    modelo_chasis VARCHAR(30) COLLATE utf8mb4_spanish_ci NOT NULL,
    marca_chasis VARCHAR(30) COLLATE utf8mb4_spanish_ci NOT NULL,
    web_marca_chasis VARCHAR(100) COLLATE utf8mb4_spanish_ci,
    foto_chasis VARCHAR(70) COLLATE utf8mb4_spanish_ci,
    homologacion_chasis VARCHAR(70) COLLATE utf8mb4_spanish_ci)  ENGINE=InnoDB  DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_spanish_ci;

CREATE TABLE motores (
    id_motor INT UNSIGNED NOT NULL AUTO_INCREMENT PRIMARY KEY,
    modelo_motor VARCHAR(30) COLLATE utf8mb4_spanish_ci NOT NULL,
    marca_motor VARCHAR(30) COLLATE utf8mb4_spanish_ci NOT NULL,
    web_marca_motores VARCHAR(100) COLLATE utf8mb4_spanish_ci,
    foto_motor VARCHAR(70) COLLATE utf8mb4_spanish_ci,
    homologacion_motor VARCHAR(70) COLLATE utf8mb4_spanish_ci) ENGINE=InnoDB  DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_spanish_ci;

CREATE TABLE pilotos (
    id_piloto INT UNSIGNED NOT NULL AUTO_INCREMENT PRIMARY KEY,
    nombre_piloto VARCHAR(30) COLLATE utf8mb4_spanish_ci NOT NULL,
    apellido_piloto VARCHAR(50) COLLATE utf8mb4_spanish_ci,
    fecha_nacimiento DATE,
    web_piloto VARCHAR(100) COLLATE utf8mb4_spanish_ci,
    email_piloto VARCHAR(60) COLLATE utf8mb4_spanish_ci,
    foto_piloto VARCHAR(70) COLLATE utf8mb4_spanish_ci,
    region_id INT UNSIGNED NOT NULL,
    escuderia_id INT UNSIGNED NOT NULL,
    patrocinador_id INT UNSIGNED NOT NULL,
    CONSTRAINT fk_region FOREIGN KEY (region_id) REFERENCES regiones(id_region),
    CONSTRAINT fk_escuderia FOREIGN KEY (escuderia_id) REFERENCES escuderias(id_escuderia),
    CONSTRAINT fk_patrocinador FOREIGN KEY (patrocinador_id) REFERENCES patrocinadores(id_patrocinador)
    ) ENGINE=InnoDB  DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_spanish_ci;

CREATE TABLE campeonatos (
    id_cto INT UNSIGNED NOT NULL AUTO_INCREMENT PRIMARY KEY,
    nombre_cto VARCHAR(50) COLLATE utf8mb4_spanish_ci NOT NULL) ENGINE=InnoDB  DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_spanish_ci;

CREATE TABLE circuitos (
    id_circuito INT UNSIGNED NOT NULL AUTO_INCREMENT PRIMARY KEY,
    nombre_circuito VARCHAR(50) COLLATE utf8mb4_spanish_ci NOT NULL,
    provincia_circuito ENUM(
    'Alava','Albacete','Alicante','Almeria','Asturias','Avila','Badajoz','Barcelona','Burgos','Caceres','Cadiz','Cantabria','Castellon','Ciudad Real','Cordoba',
    'Cuenca','Girona','Granada','Guadalajara','Gipuzkoa','Huelva','Huesca','Islas Baleares','Jaen','A Coruña','La Rioja','Las Palmas','Leon','Lleida','Lugo','Madrid','Malaga','Murcia',
    'Navarra','Orense','Palencia','Pontevedra','Salamanca','Santa Cruz de Tenerife','Segovia','Sevilla','Soria','Tarragona','Teruel','Toledo','Valencia','Valladolid','Bizkaia','Zamora','Zaragoza') COLLATE utf8mb4_spanish_ci NOT NULL,
    web_circuito VARCHAR(100) COLLATE utf8mb4_spanish_ci,
    direccion_circuito VARCHAR(60) COLLATE utf8mb4_spanish_ci,
    telefono_circuito VARCHAR(15) COLLATE utf8mb4_spanish_ci,
    localidad VARCHAR(30) COLLATE utf8mb4_spanish_ci NOT NULL) ENGINE=InnoDB  DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_spanish_ci;

CREATE TABLE carreras (
    id_carrera INT UNSIGNED NOT NULL AUTO_INCREMENT PRIMARY KEY,
    fecha_carrera DATE NOT NULL,
    dia ENUM('Despejado','Parcialmente nublado','Nublado') COLLATE utf8mb4_spanish_ci,
    temperatura TINYINT,
    humedad TINYINT UNSIGNED,
    presion SMALLINT UNSIGNED,
    viento TINYINT UNSIGNED,
    tipo_carrera ENUM('Clasificacion','Carrera 1','Carrera 2','Warm Up','Test') COLLATE utf8mb4_spanish_ci NOT NULL,
    circuito_id INT UNSIGNED NOT NULL,
    FOREIGN KEY (circuito_id) REFERENCES circuitos(id_circuito)) ENGINE=InnoDB  DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_spanish_ci;

CREATE TABLE carreras_pilotos (
    carrera_id INT UNSIGNED NOT NULL,
    piloto_id INT UNSIGNED NOT NULL,
    cto_id INT UNSIGNED NOT NULL,
    chasis_id INT UNSIGNED,
    motor_id INT UNSIGNED,
    categoria ENUM('Senior','Pre-Mini 64 Aire','Pre-Mini 60 Agua','Mini 85 Aire','Mini 60 Agua','Mini Rotax','Junior','Rotax DD2','KZ','KZ2','Tillotson T4') COLLATE utf8mb4_spanish_ci NOT NULL,
    dorsal SMALLINT,
    tiempo_total TIME(3),
    mejor_vuelta TIME(3),
    num_vueltas TINYINT UNSIGNED,
    posicion TINYINT UNSIGNED,
    puntos TINYINT UNSIGNED,
    FOREIGN KEY (carrera_id) REFERENCES carreras(id_carrera),
    FOREIGN KEY (piloto_id) REFERENCES pilotos(id_piloto),
    FOREIGN KEY (cto_id) REFERENCES campeonatos(id_cto),
    FOREIGN KEY (chasis_id) REFERENCES chasis(id_chasis),
    FOREIGN KEY (motor_id) REFERENCES motores(id_motor),
    PRIMARY KEY (carrera_id, piloto_id,cto_id,categoria)) ENGINE=InnoDB  DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_spanish_ci;

CREATE TABLE calendario (
    id_evento INT UNSIGNED NOT NULL AUTO_INCREMENT PRIMARY KEY,
    campeonato_id INT UNSIGNED NOT NULL,
    circuito_id INT UNSIGNED NOT NULL,
    fecha_evento DATE NOT NULL,
    CONSTRAINT fk_cal_cto FOREIGN KEY (campeonato_id) REFERENCES campeonatos(id_cto),
    CONSTRAINT fk_cal_cir FOREIGN KEY (circuito_id) REFERENCES circuitos(id_circuito)
);
