CREATE VIEW vista_pilotos_regiones AS
SELECT p.nombre_piloto, p.apellido_piloto, r.id_region, r.nombre_region
FROM pilotos p
JOIN regiones r ON p.region_id = r.id_region;

CREATE VIEW vista_pilotos_escuderias AS
SELECT p.nombre_piloto, p.apellido_piloto, e.id_escuderia, e.nombre_escuderia
FROM pilotos p
JOIN escuderias e ON p.escuderia_id = e.id_escuderia;

CREATE VIEW vista_pilotos_patrocinadores AS
SELECT p.nombre_piloto, p.apellido_piloto, pa.id_patrocinador, pa.nombre_patrocinador
FROM pilotos p
JOIN patrocinadores pa ON p.patrocinador_id = pa.id_patrocinador;

CREATE VIEW vista_pilotos AS
SELECT p.id_piloto, p.nombre_piloto, p.apellido_piloto, p.fecha_nacimiento, p.web_piloto, p.email_piloto, p.foto_piloto, r.id_region, r.nombre_region, e.id_escuderia, e.nombre_escuderia, pa.id_patrocinador, pa.nombre_patrocinador
FROM pilotos p
JOIN regiones r ON p.region_id = r.id_region
JOIN escuderias e ON p.escuderia_id = e.id_escuderia
JOIN patrocinadores pa ON p.patrocinador_id = pa.id_patrocinador;

CREATE VIEW vista_carreras AS
SELECT ca.id_carrera, ca.fecha_carrera, ca.dia, ca.num_vueltas, ca.pista, ca.temperatura, ca.humedad, ca.presion, ca.viento, ca.orientacion, ca.tasfalto, ct.id_carrera_tipo, ct.nombre_carrera_tipo, ci.id_circuito, ci.nombre_circuito, cp.id_cto, cp.nombre_cto
FROM carreras ca
JOIN carreras_tipo ct ON ca.carrera_tipo_id = ct.id_carrera_tipo
JOIN circuitos ci ON ca.circuito_id = ci.id_circuito
LEFT JOIN campeonatos_carreras cc ON ca.id_carrera = cc.carrera_id
LEFT JOIN campeonatos cp ON cc.cto_id = cp.id_cto;

CREATE VIEW vista_chasis AS
SELECT id_chasis, modelo_chasis, marca_chasis, web_marca_chasis, foto_chasis, ficha_tecnica_chasis
FROM chasis;

CREATE VIEW vista_regiones AS
SELECT id_region, nombre_region
FROM regiones;

CREATE VIEW vista_motores AS
SELECT id_motor, modelo_motor, marca_motor, web_marca_motores, foto_motor, ficha_tecnica_motor
FROM motores;

CREATE VIEW vista_escuderias AS
SELECT id_escuderia, nombre_escuderia, web_escuderia
FROM escuderias;

CREATE VIEW vista_patrocinadores AS
SELECT id_patrocinador, nombre_patrocinador, web_patrocinador
FROM patrocinadores;

CREATE VIEW vista_campeonatos_categorias AS
SELECT caca.id_cto, cam.nombre_cto, cat.nombre_categoria
FROM campeonatos_categorias caca
JOIN categorias cat ON caca.id_categoria = cat.id_categoria
JOIN campeonatos cam ON caca.id_cto = cam.id_cto;

CREATE VIEW vista_circuitos AS
SELECT c.id_circuito, c.nombre_circuito, c.web_circuito, c.direccion_circuito, c.localidad_circuito, c.telefono_circuito, c.altitud, c.longitud, c.curvasizd, c.curvasdcha, c.velocidadmax, c.silueta, a.nombre_area
FROM circuitos c
JOIN areas a ON c.area_id = a.id_area;

CREATE VIEW vista_resultados AS
SELECT
    r.id_resultado,
    r.carrera_id,
    r.categoria_id,
    vcar.fecha_carrera,
    vcar.id_carrera_tipo,
    vcar.nombre_carrera_tipo,
    cat.id_categoria,
    cat.nombre_categoria,
    vcar.id_circuito,
    vcar.nombre_circuito,
    p.id_piloto,
    p.nombre_piloto,
    p.apellido_piloto,
    p.foto_piloto,
    r.dorsal,
    vcar.id_cto,
    vcar.nombre_cto,
    r.tiempo_total,
    r.mejor_vuelta,
    r.num_vueltas_completadas,
    vcar.num_vueltas,
    r.posicion,
    r.comentario_posicion,
    -- LÓGICA DE PROTECCIÓN DE PUNTOS
    CASE 
        WHEN r.comentario_posicion IN ('DNS', 'DSQ') THEN 0
        WHEN r.comentario_posicion = 'DNF' THEN
            CASE 
                WHEN vcar.id_cto = 1 THEN IFNULL(sp.puntos_asignados, 0) -- Campeonato 1: SÍ puntúa
                WHEN vcar.id_cto = 2 THEN 0                             -- Campeonato 2: NO puntúa
                ELSE IFNULL(sp.puntos_asignados, 0)                     -- Otros: Por defecto sí (o cámbialo a 0)
            END
        ELSE IFNULL(sp.puntos_asignados, 0)
    END AS puntos,
    ch.id_chasis,
    ch.marca_chasis,
    ch.modelo_chasis,
    mo.id_motor,
    mo.marca_motor,
    mo.modelo_motor
FROM resultados r
JOIN pilotos p ON r.piloto_id = p.id_piloto
JOIN vista_carreras vcar ON r.carrera_id = vcar.id_carrera
JOIN categorias cat ON r.categoria_id = cat.id_categoria
LEFT JOIN chasis ch ON r.chasis_id = ch.id_chasis
LEFT JOIN motores mo ON r.motor_id = mo.id_motor
LEFT JOIN sistemas_puntuacion sp ON vcar.id_cto = sp.id_cto
                                 AND vcar.id_carrera_tipo = sp.id_tipo_carrera
                                 AND r.posicion = sp.posicion;

CREATE OR REPLACE VIEW vista_ranking AS
SELECT
    id_cto,
    nombre_cto,
    nombre_categoria,
    id_piloto,
    nombre_piloto,
    apellido_piloto,
    SUM(puntos) AS puntos_totales,
    COUNT(DISTINCT carrera_id) AS carreras_disputadas,
    -- Calculamos cuántas victorias (P1) tiene cada uno para desempates
    SUM(CASE WHEN posicion = 1 AND nombre_carrera_tipo LIKE 'Carrera%' THEN 1 ELSE 0 END) AS victorias,
    -- Contamos cuántas veces ha subido al cajón (1º, 2º o 3º)
    SUM(CASE WHEN posicion <= 3 AND nombre_carrera_tipo LIKE 'Carrera%' THEN 1 ELSE 0 END) AS podios,
    MIN(posicion) AS mejor_resultado
FROM vista_resultados
GROUP BY
    id_cto,
    nombre_cto,
    nombre_categoria,
    id_piloto,
    nombre_piloto,
    apellido_piloto
ORDER BY id_cto, puntos_totales DESC;
