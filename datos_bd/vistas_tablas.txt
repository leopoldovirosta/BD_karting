CREATE VIEW vista_pilotos_regiones AS
SELECT p.nombre_piloto, p.apellido_piloto, r.id_region, r.nombre_pais
FROM pilotos p
JOIN paises r ON p.id_pais = r.id_pais;

CREATE VIEW vista_pilotos_escuderias AS
SELECT p.nombre_piloto, p.apellido_piloto, e.id_escuderia, e.nombre_escuderia
FROM pilotos p
JOIN escuderias e ON p.id_escuderia = e.id_escuderia;

CREATE VIEW vista_pilotos_patrocinadores AS
SELECT p.nombre_piloto, p.apellido_piloto, pa.id_patrocinador, pa.nombre_patrocinador
FROM pilotos p
JOIN patrocinadores pa ON p.id_patrocinador = pa.id_patrocinador;

CREATE VIEW vista_pilotos AS
SELECT p.id_piloto, p.nombre_piloto, p.apellido_piloto, p.fecha_nacimiento, p.web_piloto, p.email_piloto, p.foto_piloto, r.id_pais, r.nombre_pais, e.id_escuderia, e.nombre_escuderia, pa.id_patrocinador, pa.nombre_patrocinador
FROM pilotos p
JOIN paises r ON p.id_pais = r.id_pais
JOIN escuderias e ON p.id_escuderia = e.id_escuderia
JOIN patrocinadores pa ON p.id_patrocinador = pa.id_patrocinador;

CREATE VIEW vista_carreras AS
SELECT ca.id_carrera, ca.fecha_carrera, ca.dia, ca.num_vueltas, ca.pista, ca.temperatura, ca.humedad, ca.presion, ca.viento, ca.orientacion, ca.tasfalto, ct.id_carrera_tipo, ct.nombre_carrera_tipo, ci.id_circuito, ci.nombre_circuito,ci.logo_circuito, cp.id_cto, cp.nombre_cto
FROM carreras ca
JOIN carreras_tipo ct ON ca.id_carrera_tipo = ct.id_carrera_tipo
JOIN circuitos ci ON ca.id_circuito = ci.id_circuito
LEFT JOIN campeonatos_carreras cc ON ca.id_carrera = cc.id_carrera
LEFT JOIN campeonatos cp ON cc.id_cto = cp.id_cto;

CREATE VIEW vista_chasis AS
SELECT ch.id_chasis, ch.id_marca, m.nombre_marca, m.logo_marca, ch.modelo_chasis, ch.tubo_diametro, ch.distancia_ejes, ch.eje_trasero, ch.sistema_frenado, ch.categoria_objetivo, ch.ano_homologacion, ch.foto_chasis
FROM chasis ch
JOIN marcas m ON ch.id_marca = m.id_marca;

CREATE VIEW vista_paises AS
SELECT id_pais, nombre_pais
FROM paises;

CREATE VIEW vista_motores AS
SELECT mo.id_motor, mo.id_marca, m.nombre_marca, m.logo_marca, mo.modelo_motor, mo.cilindrada, mo.ciclo, mo.potencia_cv, mo.refrigeracion, mo.admision, mo.num_marchas, mo.regimen_max_rpm, mo.ano_homologacion, mo.peso_motor, mo.foto_motor
FROM motores mo
JOIN marcas m ON mo.id_marca = m.id_marca;

CREATE VIEW vista_marcas AS
SELECT m.id_marca, m.nombre_marca, pa.nombre_pais, m.logo_marca, m.web_marca, m.es_marca_chasis, m.es_marca_motor, m.es_marca_ruedas
FROM marcas m
JOIN paises pa ON m.id_pais = pa.id_pais;

CREATE VIEW vista_escuderias AS
SELECT id_escuderia, nombre_escuderia, web_escuderia
FROM escuderias;

CREATE VIEW vista_patrocinadores AS
SELECT id_patrocinador, nombre_patrocinador, web_patrocinador
FROM patrocinadores;

CREATE VIEW vista_campeonatos_categorias AS
SELECT caca.id_cto, cam.nombre_cto, cat.nombre_categoria
FROM campeonatos_categorias caca
JOIN categorias cat ON caca.id_categoria = cat.id_categoria
JOIN campeonatos cam ON caca.id_cto = cam.id_cto;

CREATE VIEW vista_circuitos AS
SELECT c.id_circuito, c.nombre_circuito, c.web_circuito, c.direccion_circuito, c.localidad_circuito, c.telefono_circuito, c.altitud, c.longitud, c.curvasizd, c.curvasdcha, c.velocidadmax, c.silueta, pa.nombre_pais
FROM circuitos c
JOIN paises pa ON c.id_pais = pa.id_pais;

CREATE VIEW vista_resultados AS
SELECT
    r.id_resultado,
    r.id_carrera,
    r.id_categoria,
    vcar.fecha_carrera,
    vcar.id_carrera_tipo,
    vcar.nombre_carrera_tipo,
    cat.id_categoria,
    cat.nombre_categoria,
    vcar.id_circuito,
    vcar.nombre_circuito,
    vcar.logo_circuito,
    p.id_piloto,
    p.nombre_piloto,
    p.apellido_piloto,
    p.foto_piloto,
    r.dorsal,
    vcar.id_cto,
    vcar.nombre_cto,
    r.tiempo_total,
    r.mejor_vuelta,
    r.num_vueltas_completadas,
    vcar.num_vueltas,
    r.posicion,
    r.comentario_posicion,
    -- LÓGICA DE PROTECCIÓN DE PUNTOS
    CASE 
        WHEN r.comentario_posicion IN ('DNS', 'DSQ') THEN 0
        WHEN r.comentario_posicion = 'DNF' THEN
            CASE 
                WHEN vcar.id_cto = 1 THEN IFNULL(sp.puntos_asignados, 0) -- Campeonato 1: SÍ puntúa
                WHEN vcar.id_cto = 2 THEN 0                             -- Campeonato 2: NO puntúa
                ELSE IFNULL(sp.puntos_asignados, 0)                     -- Otros: Por defecto sí (o cámbialo a 0)
            END
        ELSE IFNULL(sp.puntos_asignados, 0)
    END AS puntos,

    -- DATOS DE CHASIS Y SU MARCA
    ch.modelo_chasis,
    m_ch.nombre_marca AS marca_chasis,
    m_ch.logo_marca AS logo_chasis,

    -- DATOS DE MOTOR Y SU MARCA
    mo.modelo_motor,
    m_mo.nombre_marca AS marca_motor,
    m_mo.logo_marca AS logo_motor,
    
    -- DATOS DE RUEDA Y SU MARCA
    ru.modelo_motor,
    m_ru.nombre_marca AS marca_rueda,
    m_ru.logo_marca AS logo_rueda
    
FROM resultados r
JOIN pilotos p ON r.id_piloto = p.id_piloto
JOIN vista_carreras vcar ON r.id_carrera = vcar.id_carrera
JOIN categorias cat ON r.id_categoria = cat.id_categoria
LEFT JOIN chasis ch ON r.id_chasis = ch.id_chasis
-- Unimos la marca del chasis usando el alias 'm_ch'
LEFT JOIN marcas m_ch ON ch.id_marca = m_ch.id_marca                    
LEFT JOIN motores mo ON r.id_motor = mo.id_motor
-- Unimos la marca del motor usando el alias 'm_mo'
LEFT JOIN marcas m_mo ON mo.id_marca = m_mo.id_marca
LEFT JOIN ruedas ru ON r.id_rueda = ru.id_rueda
-- Unimos la marca de la rueda usando el alias 'm_ru'
LEFT JOIN marcas m_ru ON ru.id_marca = m_ru.id_marca
LEFT JOIN sistemas_puntuacion sp ON vcar.id_cto = sp.id_cto
                                 AND vcar.id_carrera_tipo = sp.id_tipo_carrera
                                 AND r.posicion = sp.posicion;

CREATE OR REPLACE VIEW vista_ranking AS
SELECT
    id_cto,
    nombre_cto,
    nombre_categoria,
    id_piloto,
    nombre_piloto,
    apellido_piloto,
    SUM(puntos) AS puntos_totales,
    COUNT(DISTINCT carrera_id) AS carreras_disputadas,
    -- Calculamos cuántas victorias (P1) tiene cada uno para desempates
    SUM(CASE WHEN posicion = 1 AND nombre_carrera_tipo LIKE 'Carrera%' THEN 1 ELSE 0 END) AS victorias,
    -- Contamos cuántas veces ha subido al cajón (1º, 2º o 3º)
    SUM(CASE WHEN posicion <= 3 AND nombre_carrera_tipo LIKE 'Carrera%' THEN 1 ELSE 0 END) AS podios,
    MIN(posicion) AS mejor_resultado
FROM vista_resultados
GROUP BY
    id_cto,
    nombre_cto,
    nombre_categoria,
    id_piloto,
    nombre_piloto,
    apellido_piloto
ORDER BY id_cto, puntos_totales DESC;
